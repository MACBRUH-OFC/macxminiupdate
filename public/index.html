async function selectFile(item, element) { cleanPreviousMediaTracks(); currentPreviewRequestId++; const requestId = currentPreviewRequestId; if (element) { document.querySelectorAll('.sub-asset-row').forEach(el => el.classList.remove('bg-zinc-900', 'text-white')); element.classList.add('bg-zinc-900', 'text-white'); } document.getElementById('previewFallback').classList.add('hidden'); document.getElementById('previewContent').classList.remove('hidden'); const mediaBox = document.getElementById('mediaBox'); const mediaTitle = document.getElementById('mediaTitle'); const mediaTypeLabel = document.getElementById('mediaTypeLabel'); const saveBtn = document.getElementById('singleSaveBtn'); mediaTitle.innerText = item.name || item.fileName || "Loading..."; mediaTypeLabel.innerText = item.label || "Processing..."; // Show loading spinner mediaBox.innerHTML = `

Extracting bundle contents...

`; document.getElementById('previewTargetWorkspace').scrollIntoView({ behavior: 'smooth', block: 'nearest' }); try { // Call /api/process to extract the bundle const processUrl = `/api/process?bundle_url=${encodeURIComponent(item.id || item.url)}`; const processResponse = await fetch(processUrl); if (!processResponse.ok) { if (requestId === currentPreviewRequestId) { showFallbackError(`Failed to extract: HTTP ${processResponse.status}`); } return; } const extractedData = await processResponse.json(); if (requestId !== currentPreviewRequestId) return; if (!extractedData.success) { showFallbackError(extractedData.message || "Failed to extract bundle"); return; } // Cache the extracted data offlineDataPayloadSessionCache[item.id || item.url] = extractedData; // Display the extracted categories if (extractedData.categories && Object.keys(extractedData.categories).length > 0) { displayExtractedAssets(extractedData, requestId); } else if (extractedData.composites && extractedData.composites.length > 0) { // Show composites if available displayComposites(extractedData.composites, requestId); } else { showFallbackError("No assets extracted from bundle"); } } catch (error) { if (requestId === currentPreviewRequestId) { showFallbackError(`Error: ${error.message}`); } console.error('selectFile error:', error); } } function displayExtractedAssets(extractedData, requestId) { if (requestId !== currentPreviewRequestId) return; const mediaBox = document.getElementById('mediaBox'); let html = `
`; // Display composites if (extractedData.composites && extractedData.composites.length > 0) { html += `
🎨 3D Assemblies
`; extractedData.composites.forEach(comp => { html += `
${comp.name}
${comp.type}
`; }); html += `
`; } // Display asset categories const categories = [ { key: "3d_models", label: "3D Models", icon: "fa-cube" }, { key: "textures", label: "Textures", icon: "fa-image" }, { key: "materials", label: "Materials", icon: "fa-wand-magic-sparkles" }, { key: "animations", label: "Animations", icon: "fa-person-running" }, { key: "configs", label: "Configs", icon: "fa-gear" }, { key: "others", label: "Others", icon: "fa-file" } ]; categories.forEach(cat => { const assets = extractedData.categories[cat.key] || []; if (assets.length === 0) return; html += `
${cat.label} (${assets.length})
`; assets.forEach((asset, idx) => { const fname = asset.fileName || asset.name || `asset_${idx}`; const shortName = fname.split('/').pop().substring(0, 30); if (cat.key === 'textures') { html += `
${shortName}
`; } else { html += `
${shortName}
`; } }); }); html += `
`; document.getElementById('mediaBox').innerHTML = html; } function displayComposites(composites, requestId) { if (requestId !== currentPreviewRequestId) return; const mediaBox = document.getElementById('mediaBox'); let html = `
`; composites.forEach(comp => { html += `
${comp.name}
${comp.type}
${comp.preview_summary}
`; }); html += `
`; document.getElementById('mediaBox').innerHTML = html; } function showFallbackError(message) { const mediaBox = document.getElementById('mediaBox'); mediaBox.innerHTML = `

${message}

`; }
